Table role {
  role_id int [pk, increment]
  name varchar(50) [not null, unique]
}

Table user {
  user_id int [pk, increment]
  name varchar(100) [not null]
  email varchar(150) [not null, unique]
  password_hash varchar(255) [not null]
  role_id int [not null]
  created_at timestamp [not null]
  updated_at timestamp [not null]
  is_active boolean [not null]
}

Table player {
  player_id int [pk, increment]
  user_id int [not null, unique]
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Table deck {
  deck_id int [pk, increment]
  player_id int [not null]
  name varchar(100) [not null]
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Table card {
  card_id int [pk, increment]
  card_name varchar(100) [not null]
  value int [not null]
  type varchar(50) [not null]
  effect varchar(255)
  max_quantity int [not null]
  rarity varchar(50) [not null]
  description varchar(255)
}

Table deck_card {
  deck_id int [not null]
  card_id int [not null]
  quantity int [not null]

  indexes {
    (deck_id, card_id) [pk]
  }
}

Table tournament {
  tournament_id int [pk, increment]
  name varchar(100) [not null]
  date date [not null]
  format varchar(50) [not null]
  num_players int [not null]
  status varchar(50) [not null]
  organizer_id int [not null]
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Table round {
  round_id int [pk, increment]
  tournament_id int [not null]
  number int [not null]
  name varchar(100) [not null]
  status varchar(50) [not null]
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Table tournament_registration {
  tournament_registration_id int [pk, increment]
  tournament_id int [not null]
  player_id int [not null]
  registered_at timestamp [not null]
}

Table round_deck_selection {
  round_deck_selection_id int [pk, increment]
  round_id int [not null]
  player_id int [not null]
  deck_id int [not null]
  selected_at timestamp [not null]
}

Table match {
  match_id int [pk, increment]
  round_id int [not null]
  deck1_id int [not null]
  deck2_id int [not null]
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Table match_deck_card {
  match_deck_card_id int [pk, increment]
  match_id int [not null]
  card_id int [not null]
  player_slot int [not null]
  state varchar(50) [not null]
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Table match_player_state {
  match_id int [not null]
  player_slot int [not null]
  current_life int [not null]
  current_shield int [not null]
  created_at timestamp [not null]
  updated_at timestamp [not null]

  indexes {
    (match_id, player_slot) [pk]
  }
}

Table result {
  result_id int [pk, increment]
  match_id int [not null]
  winner_id int
  end_phase varchar(50)
  player1_final_life int
  player2_final_life int
  player1_points int
  player2_points int
  date timestamp
  created_at timestamp [not null]
}

Table audit_log {
  audit_id int [pk, increment]
  user_id int [not null]
  action varchar(50) [not null]
  table_name varchar(100) [not null]
  record_id int
  old_values json
  new_values json
  created_at timestamp [not null]
}
